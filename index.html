// --- RENDERIZADO RESPONSIVO (TABLA EN DESKTOPS / TARJETAS EN CELULARES) --- function renderizarRegistrosResponsivo(resultados) { const tbody = document.getElementById('tablaRegistros'); if (!tbody) return; tbody.innerHTML = ''; if (resultados.length === 0) { tbody.innerHTML = ` No se encontraron registros que coincidan con los filtros aplicados. `; return; } resultados.forEach(item => { const precioPre = Number(item.precioPre) || 0; const precioVenta = Number(item.precioVenta) || 0; const margenPesos = precioVenta - precioPre; const margenPorc = (precioPre > 0) ? ((margenPesos / precioPre) * 100).toFixed(2) : "0.00"; let totalPdfs = (item.archivosPdf && item.archivosPdf.length > 0) ? item.archivosPdf.length : 0; let badgePdf = totalPdfs > 0 ? `${totalPdfs}` : 'Sin PDF'; // ------------------------------------------------------------- // VISTA PARA COMPUTADORAS Y TABLETS LARGAS (FILA DE TABLA CLÁSICA) // ------------------------------------------------------------- let filaEscritorio = `
${item.id}
${item.idEnlace || 'S/N'}
${item.cartera || 'GENERAL'}
${item.credito || 'S/N'}
${item.direccion || 'S/D'}
${item.colonia || 'S/D'}
$${precioPre.toLocaleString('es-MX', {minimumFractionDigits: 2})} $${precioVenta.toLocaleString('es-MX', {minimumFractionDigits: 2})} $${margenPesos.toLocaleString('es-MX', {minimumFractionDigits: 2})} (${margenPorc}%) ${item.etapaActual || item.etapa || 'S/D'} ${badgePdf} ${item.responsable || 'S/N'}
`; // ------------------------------------------------------------- // VISTA PARA CELULARES (TARJETA COMPACTA VERTICAL TIPO APP MÓVIL) // ------------------------------------------------------------- let tarjetaMovil = `
${item.id} ${item.idEnlace || 'S/N'}
${badgePdf}

${item.direccion || 'S/D'}

${item.colonia || 'S/D'}, ${item.municipio || ''}

Precio Pre-Auto $${precioPre.toLocaleString('es-MX', {minimumFractionDigits: 2})}
Precio Venta $${precioVenta.toLocaleString('es-MX', {minimumFractionDigits: 2})}
Etapa Actual ${item.etapaActual || item.etapa || 'S/D'}
Responsable ${item.responsable || 'S/N'}
`; tbody.innerHTML += filaEscritorio + tarjetaMovil; }); }
